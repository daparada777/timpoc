# timpoc — зависимости движка распознавания.
#
# ВАЖНО ПРО av<14 (найдено реальной ошибкой, не предположение):
#   faster-whisper 1.2.1 вызывает av.open(..., metadata_errors="ignore"),
#   а в PyAV 14+ этот параметр удалён. С av 19.0.1 распознавание падает на
#   первой же строке:
#       TypeError: open() got an unexpected keyword argument 'metadata_errors'
#   Поэтому av закреплён. Проверено: с av 13.1.0 распознавание работает.
#   Если когда-нибудь faster-whisper починит вызов — ограничение можно снять.
#
# PyAV содержит FFmpeg внутри себя, поэтому пользователю НЕ нужно устанавливать
# FFmpeg отдельно. Это осознанное решение: меньше шагов установки и меньше
# поводов для проблем с антивирусами.

faster-whisper>=1.2.1,<2
av>=11,<14
numpy>=1.24

# huggingface-hub ставится как зависимость faster-whisper; фиксируем минимум,
